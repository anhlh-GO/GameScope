import React from "react";

export function MiniBar({ value = 0, max = 1, color = "var(--accent)", width = 64, height = 6, label, style }) {
  const f = max > 0 ? Math.max(0, Math.min(1, value / max)) : 0;
  return (
    <span style={{ display: "inline-flex", alignItems: "center", gap: 8, justifyContent: "flex-end", ...style }}>
      {label != null ? <span style={{ fontVariantNumeric: "tabular-nums" }}>{label}</span> : null}
      <span style={{ width: width, height: height, borderRadius: height, background: "var(--neutral-soft)", overflow: "hidden", flex: "none" }}>
        <span style={{ display: "block", width: (f * 100) + "%", height: "100%", borderRadius: height, background: color }}></span>
      </span>
    </span>
  );
}
