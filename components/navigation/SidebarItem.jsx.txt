import React from "react";
import { Icon } from "../core/Icon.jsx";

export function SidebarItem({ icon, label, active = false, collapsed = false, badge, onClick, href, disabled = false }) {
  const [hover, setHover] = React.useState(false);
  const Tag = href ? "a" : "button";
  return (
    <Tag href={href} onClick={disabled ? undefined : onClick} title={collapsed ? label : undefined} aria-current={active ? "page" : undefined}
      onMouseEnter={function () { setHover(true); }} onMouseLeave={function () { setHover(false); }}
      style={{ position: "relative", display: "flex", alignItems: "center", gap: 10, width: "100%", height: 32, padding: collapsed ? 0 : "0 10px",
        justifyContent: collapsed ? "center" : "flex-start", border: 0, borderRadius: "var(--radius-md)", textDecoration: "none", textAlign: "left",
        background: active ? "var(--bg-active)" : hover ? "var(--bg-hover)" : "transparent",
        color: active ? "var(--text-primary)" : hover ? "var(--text-primary)" : "var(--text-secondary)",
        font: (active ? "500" : "400") + " 13px/1 var(--font-sans)", cursor: disabled ? "not-allowed" : "pointer", opacity: disabled ? 0.45 : 1 }}>
      {active ? <span style={{ position: "absolute", left: collapsed ? 2 : -8, top: 8, bottom: 8, width: 2, borderRadius: 2, background: "var(--accent)" }}></span> : null}
      <span style={{ display: "inline-flex", color: active ? "var(--accent)" : "inherit" }}><Icon name={icon} size={17} /></span>
      {collapsed ? null : <span style={{ flex: 1, overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>{label}</span>}
      {!collapsed && badge != null ? <span style={{ font: "600 10px/1 var(--font-mono)", padding: "3px 5px", borderRadius: 4, background: "var(--negative-soft)", color: "var(--negative)" }}>{badge}</span> : null}
      {collapsed && badge != null ? <span style={{ position: "absolute", top: 6, right: 10, width: 6, height: 6, borderRadius: 3, background: "var(--negative)" }}></span> : null}
    </Tag>
  );
}
