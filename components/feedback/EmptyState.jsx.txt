import React from "react";
import { Icon } from "../core/Icon.jsx";

const KINDS = {
  empty: { icon: "inbox", tone: "var(--text-tertiary)", title: "No data for this selection" },
  error: { icon: "triangle-alert", tone: "var(--negative)", title: "Couldn't load this data" },
  "no-permission": { icon: "lock", tone: "var(--warning)", title: "You don't have access" },
  loading: { icon: "loader", tone: "var(--text-tertiary)", title: "Loading…" },
  "coming-soon": { icon: "hourglass", tone: "var(--info)", title: "Coming soon" }
};

export function EmptyState({ kind = "empty", title, description, action, compact = false, style }) {
  const k = KINDS[kind] || KINDS.empty;
  return (
    <div style={{ display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center", textAlign: "center", gap: 8, padding: compact ? "20px 16px" : "40px 24px", ...style }}>
      <span style={{ width: compact ? 32 : 40, height: compact ? 32 : 40, borderRadius: "var(--radius-lg)", display: "inline-flex", alignItems: "center", justifyContent: "center",
        background: "var(--neutral-soft)", color: k.tone }}>
        <Icon name={k.icon} size={compact ? 16 : 20} style={kind === "loading" ? { animation: "gp-spin 1.2s linear infinite" } : undefined} />
      </span>
      <div style={{ font: "600 14px/1.3 var(--font-sans)", color: "var(--text-primary)", marginTop: 4 }}>{title || k.title}</div>
      {description ? <div style={{ font: "var(--type-body)", color: "var(--text-tertiary)", maxWidth: 380, textWrap: "pretty" }}>{description}</div> : null}
      {action ? <div style={{ marginTop: 8 }}>{action}</div> : null}
    </div>
  );
}
