import React from "react";
import { Icon } from "../core/Icon.jsx";

export function Select({ options = [], value, onChange, multiple = false, label, placeholder = "Select", searchable = false, size = "md", disabled = false, disabledReason, width, menuWidth, align = "left", renderOption, style }) {
  const [open, setOpen] = React.useState(false);
  const [q, setQ] = React.useState("");
  const [hover, setHover] = React.useState(false);
  const ref = React.useRef(null);
  React.useEffect(function () {
    if (!open) return;
    function d(e) { if (ref.current && !ref.current.contains(e.target)) setOpen(false); }
    function k(e) { if (e.key === "Escape") setOpen(false); }
    document.addEventListener("mousedown", d); document.addEventListener("keydown", k);
    return function () { document.removeEventListener("mousedown", d); document.removeEventListener("keydown", k); };
  }, [open]);
  const vals = multiple ? (value || []) : [value];
  const sel = options.filter(function (o) { return vals.indexOf(o.value) >= 0; });
  let text = placeholder;
  if (sel.length === 1) text = sel[0].label;
  else if (sel.length > 1) text = sel.length + " selected";
  const shown = q ? options.filter(function (o) { return String(o.label).toLowerCase().indexOf(q.toLowerCase()) >= 0; }) : options;
  const h = { sm: 28, md: 32, lg: 40 }[size] || 32;
  function pick(o) {
    if (o.disabled) return;
    if (multiple) {
      const next = vals.indexOf(o.value) >= 0 ? vals.filter(function (v) { return v !== o.value; }) : vals.concat([o.value]);
      if (onChange) onChange(next);
    } else { if (onChange) onChange(o.value); setOpen(false); }
  }
  return (
    <div ref={ref} style={{ position: "relative", display: "inline-flex", width: width, flex: width ? undefined : "none", maxWidth: "100%", ...style }} title={disabled ? disabledReason : undefined}>
      <button type="button" disabled={disabled} onClick={function () { setOpen(!open); setQ(""); }}
        onMouseEnter={function () { setHover(true); }} onMouseLeave={function () { setHover(false); }}
        style={{ display: "inline-flex", alignItems: "center", gap: 6, height: h, padding: "0 8px 0 10px", width: width ? "100%" : "auto", minWidth: 0,
          borderRadius: "var(--radius-md)", border: "1px solid " + (open ? "var(--accent-border)" : "var(--border-default)"),
          background: hover && !disabled ? "var(--bg-active)" : "var(--bg-raised)", color: "var(--text-primary)", font: "500 " + (size === "sm" ? 12 : 13) + "px/1 var(--font-sans)",
          cursor: disabled ? "not-allowed" : "pointer", opacity: disabled ? 0.45 : 1, whiteSpace: "nowrap", textAlign: "left" }}>
        {label ? <span style={{ color: "var(--text-tertiary)", fontWeight: 400 }}>{label}</span> : null}
        {sel.length === 1 && sel[0].prefix ? sel[0].prefix : null}
        <span style={{ flex: width ? 1 : "none", minWidth: 0, overflow: "hidden", textOverflow: "ellipsis" }}>{text}</span>
        {disabled && disabledReason ? <Icon name="lock" size={12} color="var(--text-tertiary)" /> : <Icon name="chevron-down" size={14} color="var(--text-tertiary)" />}
      </button>
      {open ? (
        <div role="listbox" aria-multiselectable={multiple || undefined} style={{ position: "absolute", top: h + 4, left: align === "left" ? 0 : undefined, right: align === "right" ? 0 : undefined,
          zIndex: "var(--z-popover)", minWidth: "100%", width: menuWidth, maxHeight: 340, overflow: "auto", padding: 4, borderRadius: "var(--radius-lg)",
          background: "var(--bg-tooltip)", boxShadow: "var(--shadow-md)", animation: "gp-fade-up var(--dur-fast) var(--ease-out)" }}>
          {searchable ? (
            <div style={{ padding: 4 }}>
              <input autoFocus value={q} onChange={function (e) { setQ(e.target.value); }} placeholder="Search…"
                style={{ width: "100%", height: 28, padding: "0 8px", borderRadius: 6, border: "1px solid var(--border-default)", background: "var(--bg-sunken)", color: "var(--text-primary)", font: "400 12px/1 var(--font-sans)", outline: "none" }} />
            </div>
          ) : null}
          {shown.map(function (o) {
            const on = vals.indexOf(o.value) >= 0;
            if (o.divider) return <div key={o.value} style={{ height: 1, background: "var(--border-subtle)", margin: "4px 2px" }}></div>;
            return (
              <div key={o.value} role="option" aria-selected={on} aria-disabled={o.disabled || undefined} title={o.disabled ? o.disabledReason : undefined}
                onClick={function () { pick(o); }}
                onMouseEnter={function (e) { e.currentTarget.style.background = o.disabled ? "transparent" : "var(--bg-hover)"; }}
                onMouseLeave={function (e) { e.currentTarget.style.background = "transparent"; }}
                style={{ display: "flex", alignItems: "center", gap: 8, minHeight: 30, padding: "5px 8px", borderRadius: 6, cursor: o.disabled ? "not-allowed" : "pointer",
                  opacity: o.disabled ? 0.45 : 1, color: "var(--text-primary)", font: "400 13px/1.3 var(--font-sans)" }}>
                {multiple ? (
                  <span style={{ width: 14, height: 14, flex: "none", borderRadius: 4, display: "inline-flex", alignItems: "center", justifyContent: "center",
                    background: on ? "var(--accent)" : "transparent", border: on ? "0" : "1px solid var(--border-strong)", color: "var(--text-on-accent)" }}>{on ? <Icon name="check" size={11} strokeWidth={3} /> : null}</span>
                ) : null}
                {o.prefix || null}
                <span style={{ flex: 1, minWidth: 0 }}>
                  {renderOption ? renderOption(o) : o.label}
                  {o.description ? <span style={{ display: "block", font: "var(--type-caption)", color: "var(--text-tertiary)", marginTop: 2 }}>{o.description}</span> : null}
                </span>
                {!multiple && on ? <Icon name="check" size={14} color="var(--accent)" /> : null}
                {o.disabled ? <Icon name="lock" size={12} color="var(--text-tertiary)" /> : null}
              </div>
            );
          })}
          {shown.length === 0 ? <div style={{ padding: "10px 8px", font: "var(--type-caption)", color: "var(--text-tertiary)" }}>No matches</div> : null}
        </div>
      ) : null}
    </div>
  );
}
