import React from "react";

export function Switch({ checked = false, onChange, label, description, disabled = false, style }) {
  return (
    <label style={{ display: "flex", alignItems: "flex-start", gap: 10, cursor: disabled ? "not-allowed" : "pointer", opacity: disabled ? 0.45 : 1, ...style }}>
      <button type="button" role="switch" aria-checked={checked} disabled={disabled} onClick={function () { if (onChange) onChange(!checked); }}
        style={{ width: 30, height: 18, flex: "none", padding: 2, border: 0, borderRadius: 9, marginTop: 1, background: checked ? "var(--accent)" : "var(--border-strong)", cursor: "inherit", transition: "background var(--dur-base)" }}>
        <span style={{ display: "block", width: 14, height: 14, borderRadius: 7, background: "#fff", boxShadow: "0 1px 2px rgba(0,0,0,.3)", transform: checked ? "translateX(12px)" : "none", transition: "transform var(--dur-base) var(--ease-out)" }}></span>
      </button>
      {label || description ? (
        <span style={{ display: "flex", flexDirection: "column", gap: 2 }}>
          {label ? <span style={{ font: "500 13px/1.35 var(--font-sans)", color: "var(--text-primary)" }}>{label}</span> : null}
          {description ? <span style={{ font: "var(--type-caption)", color: "var(--text-tertiary)" }}>{description}</span> : null}
        </span>
      ) : null}
    </label>
  );
}
