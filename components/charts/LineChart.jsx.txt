import React from "react";
import { Icon } from "../core/Icon.jsx";

function useWidth(initial) {
  const ref = React.useRef(null);
  const [w, setW] = React.useState(initial || 600);
  React.useLayoutEffect(function () {
    const el = ref.current; if (!el) return;
    setW(el.clientWidth || initial || 600);
    if (typeof ResizeObserver === "undefined") return;
    const ro = new ResizeObserver(function (e) { const cw = e[0].contentRect.width; if (cw > 0) setW(cw); });
    ro.observe(el);
    return function () { ro.disconnect(); };
  }, []);
  return [ref, w];
}
function niceStep(r) { if (!(r > 0)) return 1; const e = Math.pow(10, Math.floor(Math.log10(r))); const f = r / e; return (f <= 1 ? 1 : f <= 2 ? 2 : f <= 2.5 ? 2.5 : f <= 5 ? 5 : 10) * e; }
function niceExtent(mn, mx, ticks) { if (mx === mn) { mx = mn + 1; } const st = niceStep((mx - mn) / (ticks || 4)); return [Math.floor(mn / st) * st, Math.ceil(mx / st) * st, st]; }

export function LineChart({ data = [], series = [], xKey = "x", height = 240, yFormat, yFormatRight, xFormat, showLegend = true, zoomable = true, yMin, yMax, referenceLines = [], emptyText = "No data for this selection", style }) {
  const [ref, w] = useWidth(640);
  const [hidden, setHidden] = React.useState({});
  const [zoom, setZoom] = React.useState(null);
  const [drag, setDrag] = React.useState(null);
  const [hi, setHi] = React.useState(null);
  const fy = yFormat || function (v) { return String(Math.round(v * 100) / 100); };
  const fyr = yFormatRight || fy;
  const fx = xFormat || function (v) { return v; };
  const sig = data.length + "|" + (data[0] ? data[0][xKey] : "") + "|" + (data.length ? data[data.length - 1][xKey] : "");
  React.useEffect(function () { setZoom(null); }, [sig]);
  const view = zoom ? data.slice(zoom[0], zoom[1] + 1) : data;
  const vis = series.filter(function (s) { return !hidden[s.key]; });
  const L = vis.filter(function (s) { return s.axis !== "right"; }), R = vis.filter(function (s) { return s.axis === "right"; });
  const hasR = series.some(function (s) { return s.axis === "right"; });
  const padL = 52, padR = hasR ? 52 : 14, padT = 10, padB = 26;
  const iw = Math.max(10, w - padL - padR), ih = height - padT - padB;
  function ext(ss) {
    let mn = Infinity, mx = -Infinity;
    view.forEach(function (d) { ss.forEach(function (s) { const v = d[s.key]; if (v == null || !isFinite(v)) return; if (v < mn) mn = v; if (v > mx) mx = v; }); });
    if (mx === -Infinity) return [0, 1, 0.25];
    mn = yMin != null ? yMin : Math.min(0, mn);
    if (yMax != null) mx = yMax;
    return niceExtent(mn, mx, 4);
  }
  const eL = ext(L), eR = ext(R);
  const n = view.length;
  function X(i) { return padL + (n <= 1 ? iw / 2 : i / (n - 1) * iw); }
  function Y(v, e) { return padT + ih - (v - e[0]) / (e[1] - e[0] || 1) * ih; }
  function idxAt(clientX) { const r = ref.current.getBoundingClientRect(); const px = clientX - r.left; return Math.max(0, Math.min(n - 1, Math.round((px - padL) / iw * (n - 1)))); }
  const ticks = []; for (let t = eL[0]; t <= eL[1] + 1e-9; t += eL[2]) ticks.push(t);
  const ticksR = []; if (hasR) for (let t = eR[0]; t <= eR[1] + 1e-9; t += eR[2]) ticksR.push(t);
  const every = Math.max(1, Math.ceil(n / Math.max(2, Math.floor(iw / 72))));
  function path(s) {
    const e = s.axis === "right" ? eR : eL; let d = "", pen = false;
    view.forEach(function (row, i) { const v = row[s.key]; if (v == null || !isFinite(v)) { pen = false; return; } d += (pen ? "L" : "M") + X(i).toFixed(1) + "," + Y(v, e).toFixed(1); pen = true; });
    return d;
  }
  function areaPath(s) {
    const e = s.axis === "right" ? eR : eL; const pts = [];
    view.forEach(function (row, i) { const v = row[s.key]; if (v != null && isFinite(v)) pts.push([X(i), Y(v, e)]); });
    if (pts.length < 2) return "";
    const base = Y(Math.max(e[0], 0), e);
    return "M" + pts.map(function (p) { return p[0].toFixed(1) + "," + p[1].toFixed(1); }).join("L") + "L" + pts[pts.length - 1][0] + "," + base + "L" + pts[0][0] + "," + base + "Z";
  }
  const empty = !n || !vis.length || view.every(function (d) { return vis.every(function (s) { return d[s.key] == null; }); });
  const row = hi != null ? view[hi] : null;
  return (
    <div style={{ minWidth: 0, ...style }}>
      <div ref={ref} style={{ position: "relative", height: height, userSelect: "none" }}
        onMouseMove={function (e) { if (!n) return; const i = idxAt(e.clientX); setHi(i); if (drag) setDrag({ a: drag.a, b: i }); }}
        onMouseLeave={function () { setHi(null); setDrag(null); }}
        onMouseDown={function (e) { if (zoomable && n > 4) setDrag({ a: idxAt(e.clientX), b: idxAt(e.clientX) }); }}
        onMouseUp={function () { if (drag && Math.abs(drag.a - drag.b) >= 2) { const o = zoom ? zoom[0] : 0; setZoom([o + Math.min(drag.a, drag.b), o + Math.max(drag.a, drag.b)]); } setDrag(null); }}>
        <svg width={w} height={height} style={{ display: "block", overflow: "visible", cursor: zoomable && n > 4 ? "crosshair" : "default" }}>
          {ticks.map(function (t) { return <g key={"t" + t}><line x1={padL} x2={padL + iw} y1={Y(t, eL)} y2={Y(t, eL)} stroke="var(--chart-grid)" /><text x={padL - 8} y={Y(t, eL)} dy="0.32em" textAnchor="end" fill="var(--chart-axis)" style={{ font: "400 10.5px var(--font-sans)", fontVariantNumeric: "tabular-nums" }}>{fy(t)}</text></g>; })}
          {ticksR.map(function (t) { return <text key={"r" + t} x={padL + iw + 8} y={Y(t, eR)} dy="0.32em" fill="var(--chart-axis)" style={{ font: "400 10.5px var(--font-sans)", fontVariantNumeric: "tabular-nums" }}>{fyr(t)}</text>; })}
          {view.map(function (d, i) { return i % every === 0 ? <text key={"x" + i} x={X(i)} y={height - 8} textAnchor="middle" fill="var(--chart-axis)" style={{ font: "400 10.5px var(--font-sans)" }}>{fx(d[xKey])}</text> : null; })}
          {referenceLines.map(function (r, i) { return <g key={"ref" + i}><line x1={padL} x2={padL + iw} y1={Y(r.value, eL)} y2={Y(r.value, eL)} stroke={r.color || "var(--warning)"} strokeDasharray="4 4" /><text x={padL + iw} y={Y(r.value, eL) - 5} textAnchor="end" fill={r.color || "var(--warning)"} style={{ font: "500 10.5px var(--font-sans)" }}>{r.label}</text></g>; })}
          {drag ? <rect x={X(Math.min(drag.a, drag.b))} y={padT} width={Math.abs(X(drag.b) - X(drag.a))} height={ih} fill="var(--accent-soft)" /> : null}
          {vis.map(function (s) { return s.area ? <path key={"a" + s.key} d={areaPath(s)} fill={s.color} opacity="0.12" /> : null; })}
          {vis.map(function (s) { return <path key={s.key} d={path(s)} fill="none" stroke={s.color} strokeWidth={s.dashed ? 1.4 : 1.8} strokeDasharray={s.dashed ? "4 3" : undefined} strokeLinejoin="round" strokeLinecap="round" />; })}
          {row ? <line x1={X(hi)} x2={X(hi)} y1={padT} y2={padT + ih} stroke="var(--chart-crosshair)" /> : null}
          {row ? vis.map(function (s) { const v = row[s.key]; if (v == null) return null; return <circle key={"c" + s.key} cx={X(hi)} cy={Y(v, s.axis === "right" ? eR : eL)} r="3.5" fill="var(--bg-surface)" stroke={s.color} strokeWidth="2" />; }) : null}
        </svg>
        {empty ? <div style={{ position: "absolute", inset: 0, display: "flex", alignItems: "center", justifyContent: "center", font: "var(--type-body)", color: "var(--text-tertiary)" }}>{emptyText}</div> : null}
        {row && !empty ? (
          <div style={{ position: "absolute", top: 4, left: X(hi) > w / 2 ? undefined : X(hi) + 12, right: X(hi) > w / 2 ? w - X(hi) + 12 : undefined, pointerEvents: "none", zIndex: 2,
            minWidth: 150, padding: "8px 10px", borderRadius: "var(--radius-md)", background: "var(--bg-tooltip)", boxShadow: "var(--shadow-md)", font: "400 12px/1.5 var(--font-sans)" }}>
            <div style={{ color: "var(--text-tertiary)", marginBottom: 4, fontSize: 11 }}>{fx(row[xKey], true)}</div>
            {vis.map(function (s) { const v = row[s.key]; return (
              <div key={s.key} style={{ display: "flex", alignItems: "center", gap: 8 }}>
                <span style={{ width: 8, height: s.dashed ? 2 : 8, borderRadius: 2, background: s.color }}></span>
                <span style={{ color: "var(--text-secondary)", flex: 1, whiteSpace: "nowrap" }}>{s.label}</span>
                <span style={{ color: "var(--text-primary)", fontWeight: 500, fontVariantNumeric: "tabular-nums" }}>{v == null ? "N/A" : (s.format || (s.axis === "right" ? fyr : fy))(v)}</span>
              </div>); })}
          </div>
        ) : null}
      </div>
      {showLegend && series.length > 0 || zoom ? (
        <div style={{ display: "flex", flexWrap: "wrap", alignItems: "center", gap: "6px 14px", marginTop: 10 }}>
          {showLegend ? series.map(function (s) { const off = hidden[s.key]; return (
            <button key={s.key} onClick={function () { const h = Object.assign({}, hidden); h[s.key] = !off; setHidden(h); }} title={off ? "Show series" : "Hide series"}
              style={{ display: "inline-flex", alignItems: "center", gap: 6, background: "none", border: 0, padding: 0, cursor: "pointer", font: "400 12px/1 var(--font-sans)", color: off ? "var(--text-disabled)" : "var(--text-secondary)" }}>
              <span style={{ width: 10, height: s.dashed ? 2 : 10, borderRadius: 3, background: off ? "var(--border-strong)" : s.color }}></span>{s.label}
            </button>); }) : null}
          {zoom ? <button onClick={function () { setZoom(null); }} style={{ marginLeft: "auto", display: "inline-flex", alignItems: "center", gap: 4, background: "none", border: 0, cursor: "pointer", color: "var(--accent-text)", font: "500 12px/1 var(--font-sans)" }}><Icon name="zoom-out" size={13} />Reset zoom</button> : null}
        </div>
      ) : null}
    </div>
  );
}
