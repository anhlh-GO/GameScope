import React from "react";

export function FunnelChart({ steps = [], color = "var(--accent)", format, onStepClick, showDropoff = true, style }) {
  const f = format || function (v) { return Math.round(v).toLocaleString(); };
  const first = steps.length ? steps[0].value || 1 : 1;
  const [hi, setHi] = React.useState(null);
  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 6, minWidth: 0, ...style }}>
      {steps.map(function (s, i) {
        const prev = i > 0 ? steps[i - 1].value : null;
        const conv = prev ? s.value / prev : null;
        const share = s.value / first;
        return (
          <div key={s.label + i} onMouseEnter={function () { setHi(i); }} onMouseLeave={function () { setHi(null); }} onClick={onStepClick ? function () { onStepClick(s, i); } : undefined}
            style={{ display: "grid", gridTemplateColumns: "minmax(90px,160px) minmax(0,1fr) 64px 68px", gap: 10, alignItems: "center", cursor: onStepClick ? "pointer" : "default" }}>
            <span style={{ font: "400 12px/1.25 var(--font-sans)", color: "var(--text-secondary)", overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }} title={s.label}>{s.label}</span>
            <span style={{ height: 22, borderRadius: 4, background: "var(--neutral-soft)", position: "relative", overflow: "hidden" }}>
              <span style={{ position: "absolute", left: 0, top: 0, bottom: 0, width: (share * 100) + "%", background: s.color || color, opacity: hi === i ? 1 : 0.85, borderRadius: 4, transition: "width var(--dur-slow) var(--ease-out)" }}></span>
              <span style={{ position: "absolute", left: 8, top: 0, bottom: 0, display: "flex", alignItems: "center", font: "500 11.5px/1 var(--font-sans)", color: share > 0.18 ? "var(--text-on-accent)" : "var(--text-primary)", fontVariantNumeric: "tabular-nums" }}>{f(s.value)}</span>
            </span>
            <span style={{ font: "500 12px/1 var(--font-sans)", textAlign: "right", color: "var(--text-primary)", fontVariantNumeric: "tabular-nums" }}>{(share * 100).toFixed(1)}%</span>
            <span style={{ font: "400 11.5px/1 var(--font-sans)", textAlign: "right", color: conv != null && conv < 0.6 ? "var(--negative)" : "var(--text-tertiary)", fontVariantNumeric: "tabular-nums" }}>
              {showDropoff && conv != null ? "−" + ((1 - conv) * 100).toFixed(1) + "%" : ""}
            </span>
          </div>
        );
      })}
      {steps.length ? <div style={{ display: "grid", gridTemplateColumns: "minmax(90px,160px) minmax(0,1fr) 64px 68px", gap: 10, font: "400 10.5px/1 var(--font-sans)", color: "var(--text-tertiary)", marginTop: 2 }}><span></span><span></span><span style={{ textAlign: "right" }}>of first</span><span style={{ textAlign: "right" }}>step drop</span></div> : null}
    </div>
  );
}
