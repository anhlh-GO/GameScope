import React from "react";

export function ShareBar({ segments = [], height = 8, format, showLegend = true, style }) {
  const tot = segments.reduce(function (a, s) { return a + (s.value || 0); }, 0) || 1;
  const f = format || function (v) { return Math.round(v).toLocaleString(); };
  return (
    <div style={{ minWidth: 0, ...style }}>
      <div style={{ display: "flex", gap: 2, height: height, borderRadius: height, overflow: "hidden" }}>
        {segments.map(function (s) { return <span key={s.label} title={s.label + ": " + f(s.value) + " (" + (s.value / tot * 100).toFixed(1) + "%)"} style={{ width: (s.value / tot * 100) + "%", background: s.color, minWidth: s.value > 0 ? 2 : 0 }}></span>; })}
      </div>
      {showLegend ? (
        <div style={{ display: "flex", flexWrap: "wrap", gap: "8px 18px", marginTop: 10 }}>
          {segments.map(function (s) { return (
            <div key={s.label} style={{ display: "flex", alignItems: "baseline", gap: 6, font: "400 12px/1.2 var(--font-sans)" }}>
              <span style={{ width: 8, height: 8, borderRadius: 2, background: s.color, alignSelf: "center" }}></span>
              <span style={{ color: "var(--text-secondary)" }}>{s.label}</span>
              <span style={{ color: "var(--text-primary)", fontWeight: 500, fontVariantNumeric: "tabular-nums" }}>{f(s.value)}</span>
              <span style={{ color: "var(--text-tertiary)", fontVariantNumeric: "tabular-nums" }}>{(s.value / tot * 100).toFixed(1)}%</span>
            </div>); })}
        </div>
      ) : null}
    </div>
  );
}
