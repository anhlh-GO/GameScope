import React from "react";

function useWidth(initial) {
  const ref = React.useRef(null);
  const [w, setW] = React.useState(initial || 600);
  React.useLayoutEffect(function () {
    const el = ref.current; if (!el) return;
    setW(el.clientWidth || initial || 600);
    if (typeof ResizeObserver === "undefined") return;
    const ro = new ResizeObserver(function (e) { const cw = e[0].contentRect.width; if (cw > 0) setW(cw); });
    ro.observe(el);
    return function () { ro.disconnect(); };
  }, []);
  return [ref, w];
}
function niceStep(r) { if (!(r > 0)) return 1; const e = Math.pow(10, Math.floor(Math.log10(r))); const f = r / e; return (f <= 1 ? 1 : f <= 2 ? 2 : f <= 2.5 ? 2.5 : f <= 5 ? 5 : 10) * e; }
function niceExtent(mn, mx, ticks) { if (mx === mn) { mx = mn + 1; } const st = niceStep((mx - mn) / (ticks || 4)); return [Math.floor(mn / st) * st, Math.ceil(mx / st) * st, st]; }

export function SankeyChart({ nodes = [], links = [], height = 320, nodeWidth = 10, format, style }) {
  const [ref, w] = useWidth(640);
  const [hi, setHi] = React.useState(null);
  const f = format || function (v) { return Math.round(v).toLocaleString(); };
  const cols = Math.max.apply(null, nodes.map(function (n) { return n.column; }).concat([0])) + 1;
  const byId = {}; nodes.forEach(function (n) { byId[n.id] = Object.assign({}, n, { inV: 0, outV: 0 }); });
  links.forEach(function (l) { if (byId[l.source]) byId[l.source].outV += l.value; if (byId[l.target]) byId[l.target].inV += l.value; });
  const colNodes = []; for (let c = 0; c < cols; c++) colNodes.push(nodes.filter(function (n) { return n.column === c; }).map(function (n) { return byId[n.id]; }));
  const gap = 10, padT = 4, padB = 4, labelSpace = 120;
  let k = Infinity;
  colNodes.forEach(function (ns) { const t = ns.reduce(function (a, n) { return a + Math.max(n.inV, n.outV); }, 0); const avail = height - padT - padB - gap * (ns.length - 1); if (t > 0) k = Math.min(k, avail / t); });
  if (!isFinite(k)) k = 1;
  const iw = Math.max(100, w - labelSpace);
  colNodes.forEach(function (ns, c) {
    let y = padT;
    ns.forEach(function (n) { n.x = c * (iw - nodeWidth) / Math.max(1, cols - 1); n.h = Math.max(2, Math.max(n.inV, n.outV) * k); n.y = y; n.so = 0; n.to = 0; y += n.h + gap; });
  });
  const L = links.map(function (l, i) {
    const s = byId[l.source], t = byId[l.target]; if (!s || !t) return null;
    const th = l.value * k; const y0 = s.y + s.so + th / 2; const y1 = t.y + t.to + th / 2; s.so += th; t.to += th;
    const x0 = s.x + nodeWidth, x1 = t.x, mx = (x0 + x1) / 2;
    return { i: i, l: l, th: th, d: "M" + x0 + "," + y0 + "C" + mx + "," + y0 + " " + mx + "," + y1 + " " + x1 + "," + y1, color: s.color || "var(--chart-" + ((s.column % 8) + 1) + ")" };
  }).filter(Boolean);
  const hl = hi != null ? L[hi] : null;
  return (
    <div ref={ref} style={{ position: "relative", minWidth: 0, ...style }}>
      <svg width={w} height={height} style={{ display: "block", overflow: "visible" }}>
        {L.map(function (p, i) { return <path key={i} d={p.d} fill="none" stroke={p.color} strokeWidth={Math.max(1, p.th)} strokeOpacity={hi == null ? 0.28 : hi === i ? 0.6 : 0.1} onMouseEnter={function () { setHi(i); }} onMouseLeave={function () { setHi(null); }} style={{ transition: "stroke-opacity var(--dur-fast)" }} />; })}
        {Object.keys(byId).map(function (id) { const n = byId[id]; if (n.x == null) return null; return (
          <g key={id}>
            <rect x={n.x} y={n.y} width={nodeWidth} height={n.h} rx="2" fill={n.color || "var(--chart-" + ((n.column % 8) + 1) + ")"} />
            <text x={n.x + nodeWidth + 6} y={n.y + n.h / 2} dy="0.32em" fill="var(--text-secondary)" style={{ font: "400 11px var(--font-sans)" }}>{n.label} <tspan fill="var(--text-tertiary)">{f(Math.max(n.inV, n.outV))}</tspan></text>
          </g>); })}
      </svg>
      {hl ? <div style={{ position: "absolute", top: 4, right: 4, padding: "6px 9px", borderRadius: "var(--radius-md)", background: "var(--bg-tooltip)", boxShadow: "var(--shadow-md)", font: "400 12px/1.4 var(--font-sans)", pointerEvents: "none" }}>
        <span style={{ color: "var(--text-secondary)" }}>{byId[hl.l.source].label} → {byId[hl.l.target].label}</span> <b style={{ fontVariantNumeric: "tabular-nums" }}>{f(hl.l.value)}</b>
      </div> : null}
    </div>
  );
}
