import React from "react";
import { Tooltip } from "../feedback/Tooltip.jsx";
import { Icon } from "./Icon.jsx";

export function Card({ title, subtitle, info, actions, footer, padding = 16, children, style, bodyStyle }) {
  const hasHead = title || actions;
  return (
    <section style={{ background: "var(--bg-surface)", border: "1px solid var(--border-subtle)", borderRadius: "var(--radius-xl)", display: "flex", flexDirection: "column", minWidth: 0, ...style }}>
      {hasHead ? (
        <header style={{ display: "flex", alignItems: "flex-start", gap: 12, padding: padding + "px " + padding + "px 0" }}>
          <div style={{ flex: 1, minWidth: 0 }}>
            <div style={{ display: "flex", alignItems: "center", gap: 6 }}>
              <h3 style={{ margin: 0, font: "var(--type-section)", color: "var(--text-primary)", letterSpacing: "var(--tracking-tight)" }}>{title}</h3>
              {info ? <Tooltip content={info}><span style={{ display: "inline-flex", color: "var(--text-tertiary)", cursor: "help" }}><Icon name="info" size={13} /></span></Tooltip> : null}
            </div>
            {subtitle ? <div style={{ font: "var(--type-caption)", color: "var(--text-tertiary)", marginTop: 3 }}>{subtitle}</div> : null}
          </div>
          {actions ? <div style={{ display: "flex", alignItems: "center", gap: 6, flex: "none" }}>{actions}</div> : null}
        </header>
      ) : null}
      <div style={{ padding: padding, flex: 1, minWidth: 0, ...bodyStyle }}>{children}</div>
      {footer ? <footer style={{ borderTop: "1px solid var(--border-subtle)", padding: "10px " + padding + "px", font: "var(--type-caption)", color: "var(--text-tertiary)" }}>{footer}</footer> : null}
    </section>
  );
}
